MelUi Pagination 独立分页组件

通用分页条组件：负责渲染分页控件、响应用户操作并抛出变更事件。数据请求、数据切片、列表渲染全部由业务自己控制。

【名词解释】 “钳位”一词：是开发圈技术黑话，我们文档里面出现的「页码钳位」。 含义：把一个非法 / 越界的数字，强行限制在合法的最小、最大值区间里面，不能跑出这个范围。

引入方式

<link rel="stylesheet" href="../dist/pagination/pagination.min.css">
<script src="../dist/pagination/pagination.min.js"></script>
// 两种写法都可以：
var pager = new MelUi.Pagination({ ... });   // 写法1：带 new
var pager = MelUi.Pagination({ ... });       // 写法2：省略 new，效果完全一样

配置项 Options

{
        el: null, // 挂载容器(必填)。默认null，不渲染。支持选择器字符串'#id'或DOM元素。
        enable: true, // 是否启用交互(可选)。false时所有按钮置灰禁用，DOM保留
        total: 0, // 总条数(可选)。小于0时全部交互控件置灰。
        pageSize: 20, // 每页条数(可选)。小于1自动修正为20
        pageNo: 1, // 当前页码(可选)
        pageListSize: 5, // 数字页码区域最多展示连续页码数量(可选)。最小值为3
        showTotal: true, // 是否显示总条数文本(可选)
        showListSize: true, // 是否展示中间连续数字页码(可选)。比如（1 2 3 4 5）这种
        showJumpDrop: true, // 是否显示页码下拉选择器(可选)
        showJumpInput: true, // 是否显示手动输入页码跳转框(可选)
        showPageSize: true, // 是否显示每页条数下拉(可选)
        pageSizeOptions: [10,20,50,100], // 自定义每页条数下拉选项(可选)

        // 事件
        onChange: function({pageNo, pageSize, oldPageNo, oldPageSize}) {
            // 触发场景：用户点击页码、切换每页条数、下拉选页码、输入框跳转(按回车时跳转) 都会触发
            // 说明：调用setOption更新状态不会触发onChange，避免循环
            // 参数：
            // pageNo / pageSize ：交互后的新值（页码已钳位到 1~maxPage 合法范围，业务不需要再判断越界。）
            // oldPageNo / oldPageSize ：交互前的旧值，业务据此判断是翻页还是切了每页条数
            // pageNo 当前页码， pageSize 当前每页记录数， oldPageNo 之前旧的页码， oldPageSize 之前旧的每页记录数。
        }
    }

实例 API

1. setOption(patch)：业务主动更新状态并重绘

说明：setOption 用于重绘UI，但【不会触发 onChange】，避免"setOption→onChange→setOption"死循环

局部更新：传哪些字段就更新哪些，不需要全量传入。支持的全部可更新字段：

pager.setOption({
        enable: true,              // 是否启用交互
        total: 0,                  // 总条数
        pageSize: 20,             // 每页条数
        pageNo: 1,                // 当前页码
        pageListSize: 5,          // 数字页码连续展示个数
        showTotal: true,         // 是否显示总条数文本
        showListSize: true,     // 是否显示数字页码行
        showJumpDrop: true,      // 是否显示页码下拉选择器
        showJumpInput: true,     // 是否显示手动输入页码跳转框
        showPageSize: true,      // 是否显示每页条数下拉
        pageSizeOptions: [10,20,50,100] // 每页条数下拉选项
    });

    // 说明：
    // - 只传需要改的字段即可，例如只回写总数：pager.setOption({ total: 300 });
    // - el 是挂载容器，只能在初始化时传入，setOption 不能换容器

2. destroy()：销毁组件

pager.destroy();
    // 移除事件委托、清空容器DOM；销毁后再调用任何方法都会被忽略，重复调用不报错

3. 只读属性

pager.pageNo        // 当前页码
    pager.pageSize      // 每页条数
    pager.total         // 总条数
    pager.maxPage       // 最大合法页码 = Math.max(1, Math.ceil(total / pageSize))
    pager.enable        // 是否启用
    pager.isDestroyed   // 是否已销毁

onChange 回调的三种常用写法

注意：onChange 抛出的 pageNo 永远是钳位后的合法页码（1 ~ maxPage 之间），业务不需要再判断越界。

写法A：后台标准写法 —— 切换「每页条数」时强制回到第1页

场景：用户从界面上手动更改【每页_条】 这个下拉项，即更改每页显示的记录数，要强制回到第1页时

利用回调自带的 oldPageSize 判断是否换了条数，不需要再去读外部 state：

var state = { pageNo: 1, pageSize: 20, total: 200 };

    var pager = MelUi.Pagination({
        el: '#pager',
        total: state.total,
        pageNo: state.pageNo,
        pageSize: state.pageSize,
        onChange: function(res) {
            // 对比新旧每页条数，变了就强制回第1页
            if (res.pageSize !== res.oldPageSize) {
                state.pageNo = 1;
            } else {
                state.pageNo = res.pageNo;
            }
            state.pageSize = res.pageSize;

            loadList(state.pageNo, state.pageSize); // 请求后端 / 本地切片。(请求第1页数据)
            pager.setOption(state); // 同步视图（不触发 onChange）（界面上显示在第1页）
        }
    });

写法B：保留当前页码，切条数时不强制回第1页

使用场景：用户从界面上手动更改【每页_条】 这个下拉项，即更改每页显示的记录数，不强制回第1页，比如当前是停在最后一页(比如第5页)那就停在第5页（但是因为每页记录数是用户可能改大，改大后肯定不可能有5页了，此时插件会自动修正页数，不再是第5页）

var pager = MelUi.Pagination({
        el: '#pager',
        total: 200,
        pageNo: 1,
        pageSize: 20,
        onChange: function(res) {
            // res.pageNo 已经是钳位后的合法页码，直接用
            loadList(res.pageNo, res.pageSize);
            pager.setOption({ pageNo: res.pageNo, pageSize: res.pageSize });
        }
    });

写法C：最简写法

var pager = MelUi.Pagination({
        el: '#pager',
        total: 200,
        onChange: function(res) {
            loadList(res.pageNo, res.pageSize);
        }
    });

在线示例

每个示例下方的代码块既是文档展示、也是点击"运行示例"后实际执行的代码；直接改代码块再点运行，即可看到改动效果。

示例1：基础用法（本地切片，全部默认控件）

// 先销毁旧实例，防止多次运行重复挂载
    if (window._p1 && window._p1.destroy) window._p1.destroy();

    var allData1 = [];
    for (var i = 1; i <= 137; i++) { allData1.push('数据-' + i); }

    window._p1 = MelUi.Pagination({
        el: '#demo1-pager',
        total: allData1.length,
        pageSize: 20,
        pageNo: 1,
        pageSizeOptions: [10, 20, 50],
        onChange: function(res) {
            var start = (res.pageNo - 1) * res.pageSize;
            var slice = allData1.slice(start, start + res.pageSize);
            document.getElementById('demo1-result').innerHTML = slice.join('、');
            document.getElementById('demo1-info').innerHTML =
                'pageNo=' + res.pageNo + ' ｜ pageSize=' + res.pageSize +
                ' ｜ oldPageNo=' + res.oldPageNo + ' ｜ oldPageSize=' + res.oldPageSize +
                ' ｜ maxPage=' + window._p1.maxPage;
        }
    });
    // 首次渲染
    (function(){
        var res = { pageNo: 1, pageSize: 20 };
        var start = (res.pageNo - 1) * res.pageSize;
        var slice = allData1.slice(start, start + res.pageSize);
        document.getElementById('demo1-result').innerHTML = slice.join('、');
        document.getElementById('demo1-info').innerHTML =
            'pageNo=1 ｜ pageSize=20 ｜ maxPage=' + window._p1.maxPage;
    })();

示例2：模拟远程分页（total=0 起始，setOption 回写总数）

if (window._p2 && window._p2.destroy) window._p2.destroy();

    window._p2 = MelUi.Pagination({
        el: '#demo2-pager',
        total: 0,                  // 一开始不知道总数，先传0
        pageSize: 10,
        pageNo: 1,
        showJumpDrop: false,
        pageSizeOptions: [10, 20],
        onChange: function(res) {
            document.getElementById('demo2-result').innerHTML = '加载中...';
            // 模拟 ajax 请求
            setTimeout(function() {
                var fakeTotal = 86;   // 后端返回的真实总数
                var list = [];
                for (var i = (res.pageNo - 1) * res.pageSize + 1;
                    i <= Math.min(res.pageNo * res.pageSize, fakeTotal); i++) {
                    list.push('远程数据-' + i);
                }
                // 关键：拿到总数后 setOption 回写分页条（不触发 onChange）
                window._p2.setOption({ total: fakeTotal });
                document.getElementById('demo2-result').innerHTML = list.join('、') || '（空）';
                document.getElementById('demo2-info').innerHTML =
                    'pageNo=' + res.pageNo + ' ｜ pageSize=' + res.pageSize +
                    ' ｜ total=' + fakeTotal + ' ｜ maxPage=' + window._p2.maxPage;
            }, 200);
        }
    });
    // 初始化加载第1页
    window._p2.options.onChange({ pageNo: 1, pageSize: 10, oldPageNo: 1, oldPageSize: 10 });

示例4：写法A落地 —— 切每页条数自动回到第1页

if (window._p4 && window._p4.destroy) window._p4.destroy();

    var allData4 = [];
    for (var k = 1; k <= 200; k++) { allData4.push('标准-' + k); }
    var state4 = { pageNo: 1, pageSize: 20, total: allData4.length };

    window._p4 = MelUi.Pagination({
        el: '#demo4-pager',
        total: state4.total,
        pageNo: state4.pageNo,
        pageSize: state4.pageSize,
        pageSizeOptions: [10, 20, 50],
        onChange: function(res) {
            // 切每页条数 → 强制回第1页
            if (res.pageSize !== res.oldPageSize) {
                state4.pageNo = 1;
            } else {
                state4.pageNo = res.pageNo;
            }
            state4.pageSize = res.pageSize;
            window._p4.setOption(state4);

            var start = (state4.pageNo - 1) * state4.pageSize;
            var slice = allData4.slice(start, start + state4.pageSize);
            document.getElementById('demo4-result').innerHTML = slice.join('、');
            document.getElementById('demo4-info').innerHTML =
                'pageNo=' + state4.pageNo + ' ｜ pageSize=' + state4.pageSize +
                ' ｜ maxPage=' + window._p4.maxPage + '（切每页条数时自动回到第1页）';
        }
    });
    window._p4.options.onChange({ pageNo: 1, pageSize: 20, oldPageNo: 1, oldPageSize: 20 });

MelUi Dialog 弹窗对话框组件

版本：v1.0.6 | 兼容IE9~11、移动端自适应 | MelUi.xxx / melui.xxx 大小写均可调用

一、标准引入方式


<link rel="stylesheet" href="../dist/melui.min.css">
<script src="../dist/melui.min.js"></script>

二、默认配置参数


var defaults = {
    title: "", // 弹窗标题，空则不展示标题栏
    content: "", // 弹窗正文内容
    theme: "info", // 按钮主题：info / success / warning / danger / default
    showClose: false, // 是否显示右上角关闭按钮
    btnDirection: "horizontal", // 按钮排布：horizontal横向、vert竖向
    btnFullWidth: false, // 按钮是否通栏铺满整行
    buttons: null, // 自定义按钮数组，优先级最高
    onClick: null, // 按钮回调，ret从1开始计数
    defaultValue: "" // 仅prompt生效，输入框默认文字
};
规则：仅prompt带有输入框；alert、confirm、dialog无输入框；回调参数：普通弹窗只返回按钮序号，prompt额外带回输入内容

三、调用示例

示例1：MelUi.alert 成功提示弹窗


MelUi.alert({
    title: "系统提示",
    content: "文件上传全部完成",
    theme: "success"
});

示例2：melui.confirm 确认删除（小写调用）


melui.confirm({
    title: "删除确认",
    content: "数据删除后无法恢复，确认继续吗？",
    theme: "danger",
    onClick: function(ret){
        if(ret === 2){
            alert("删除操作已确认");
        }
    }
});

示例3：MelUi.prompt 输入弹窗（唯一带输入框）


MelUi.prompt({
    title: "修改昵称",
    content: "请输入你的新昵称",
    defaultValue: "游客用户",
    onClick: function(ret, val){
        if(ret === 2){
            alert("昵称已修改为：" + val);
        }
    }
});

示例4：vert 竖向排列按钮


melui.confirm({
    title: "操作选择",
    content: "请选择需要执行的功能",
    btnDirection: "vert",
    theme: "warning"
});

示例5：按钮通栏铺满


MelUi.confirm({
    title: "设置提醒",
    content: "参数即将保存并生效",
    btnFullWidth: true
});

示例6：melui.dialog 极简快捷弹窗


melui.dialog({
    title: "系统通知",
    content: "页面配置保存完毕"
});

示例7：new 构造函数自定义多按钮弹窗


new MelUi.Dialog({
    title: "订单处理",
    content: "选择订单后续处理方式",
    showClose: true,
    btnDirection: "vert",
    btnFullWidth: true,
    buttons: [
        {text:"取消订单",type:"default"},
        {text:"申请退款",type:"danger"},
        {text:"确认收货",type:"success"}
    ],
    onClick: function(ret){
        console.log("点击了第" + ret + "个按钮");
    }
});

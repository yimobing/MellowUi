<script>
var autoStackIns = null;
// 修改容器宽度
document.getElementById('inpWrapWidth').onchange = function(){
    var val = Number(this.value);
    if(isNaN(val) || val <=0){
        val = 320;
        this.value = val;
    }
    document.getElementById('demo-auto-stack-wrap').style.width = val + 'px';
};
// 修改阈值
document.getElementById('inpThreshold').onchange = function(){
    var val = Number(this.value);
    if(isNaN(val) || val <=0){
        val = 220;
        this.value = val;
    }
    if(autoStackIns){
        autoStackIns.setThreshold(val);
    }
};
document.getElementById('btnStartStack').addEventListener('click',function(){
    var rootDom = document.getElementById('demo-auto-stack-wrap');
    var thrVal = Number(document.getElementById('inpThreshold').value) || 220;
    if(autoStackIns){
        alert('实例已启动，请勿重复创建');
        return;
    }
    autoStackIns = melui.formItemAutoStack({
        root: rootDom,
        threshold: thrVal
    });
    alert('已启动容器自动堆叠监听，当前阈值：'+ thrVal +' px');
});
document.getElementById('btnStopStack').addEventListener('click',function(){
    if(!autoStackIns){
        alert('当前没有运行中的实例');
        return;
    }
    autoStackIns.destroy();
    autoStackIns = null;
    alert('已销毁自动堆叠实例，表单项布局已还原');
});
</script>

3、工具API

melui.formItemAutoStack(options)
功能：监听容器宽度，对 mel‑form‑item‑‑search 自动增删 mel‑form‑item‑‑stack
用途：解决窄容器（弹窗 / 抽屉 / 小卡片）里面表单项被挤变形，容器空间不足时，自动将标签hd与输入框bd切换为上下两行展示。
参数 options：
‑ root：限定监听根DOM节点；不传默认 document
‑ threshold：堆叠判定阈值px，默认220
返回实例对象：{ destroy:function(), setThreshold:function(val) }
destroy()：组件销毁务必调用destroy，释放监听，并且清除stack类还原布局，防止内存泄漏
setThreshold(val)：运行时动态修改阈值
兼容：现代浏览器 ResizeObserver；IE9‑IE11降级 resize+节流定时器
注意：脚本加载不会自动执行，业务必须手动调用；支持同一个root防重复调用。

MelUi Dialog 弹窗对话框组件

版本：v1.2.1 | 兼容IE9~11、移动端自适应 | MelUi.xxx / melui.xxx 大小写均可调用

一、标准引入方式


<link rel="stylesheet" href="../src/components/dialog/index.css">
<script src="../src/components/dialog/index.js"></script>

二、默认配置参数


var defaults = {
    title: "", // 弹窗标题。空字符串不渲染标题栏。(可选)
    theme: "info", // 按钮主题。值：info/success/warning/danger/default。(可选)
    showClose: true, // 是否显示右上角关闭叉号。值：true是,false否。(可选)
    mask: true, // 是否显示遮罩层。值：true是,false否。(可选)
    maskClosable: true, // 点击遮罩是否关闭弹窗。值：true是,false否。(可选)

    btnDirection: "horizontal", // 按钮排布方向。值：horizontal横向,vert竖向。(可选)
    btnFullWidth: false, // 按钮是否通栏铺满整行。值：true是,false否。(可选)
    btnDisabled: false, // 是否一键禁用全部底部按钮。值：true是,false否。(可选)
    buttons: null, // 自定义按钮数组，优先级最高。(可选)

    message: "", // 普通文本内容。dangerouslyUseHTMLString=true时该字段无效。(可选)
    content: "", // HTML内容。仅dangerouslyUseHTMLString=true生效。(可选)
    dangerouslyUseHTMLString: false, // 是否开启HTML解析。值：true是,false否。开启后使用content字段。(可选)

    center: false, // 是否整体居中布局。值：true是,false否。true时header、body、footer全部内容水平居中。(可选)
    textCenter: false, // 是否仅正文文本居中。值：true是,false否。优先级高于center。(可选)
    fullScreen: false, // 是否全屏弹窗。值：true是,false否。(可选)
    adaptive: true, // 是否自适应内容高度。值：true是,false否。(可选)
    width: "auto", // 弹窗宽度。auto使用组件默认，支持px、百分比。(可选)
    height: "auto", // 弹窗高度。auto使用组件默认，支持px、百分比。(可选)

    device: "", // 一键移动端模式。值："phone"开启，空不开启；不自动UA检测。(可选)

    prompt: { // prompt输入框专属配置；仅调用prompt()时生效(可选)
        inputType: "input", // 输入框类型。值：input单行,textarea多行。(可选)
        placeholder: "", // 输入框占位提示文字。(可选)
        defaultValue: "", // 输入框默认值。(可选)
        unit: "" // 输入框右侧单位文字，例：元、㎡。(可选)
    },

    onClick: null, // 按钮点击回调。ret为按钮序号从1开始；prompt会附带输入框value。(可选)
    onOpen: null, // 弹窗打开完成回调。DOM已挂载完成，入参为当前实例。(可选)
    onClose: null // 弹窗关闭回调。所有关闭方式均触发；return false可阻止关闭，入参为当前实例。(可选)
};
Tip：MelUi.Tip；Toast：MelUi.Toast；onClose返回false可拦截弹窗关闭。

三、调用示例

1、极简快捷弹窗 dialog / alert，仅传message参数

melui.dialog({
    message: "保存配置完成"
});


melui.alert({
    message: "操作已执行完毕"
});


melui.alert({
    title:"测试alert自定义按钮",
    message:"alert现在支持自定义buttons",
    buttons:[
        {text:"好的我知道了",type:"success"}
    ]
});
2、Tip消息提示
调用方式说明

//（1）快速调用：直接传入字符串
// 成功 MelUi.Tip.success("消息")
// 警告 MelUi.Tip.warn("消息")
// 信息 MelUi.Tip.info("消息")
// 错误 MelUi.Tip.error("消息")

//（2）完全调用：传入object参数
MelUi.Tip({
    type: "error", // success成功 / warn警告 / info信息 / error错误
    message: "请求异常，请稍后重试", // 提示文本内容
    duration: 5000, // 自动关闭延时(毫秒)，0常驻
    position: "bottom" // 弹出位置 top / bottom
});


MelUi.Tip.success("操作执行成功");


MelUi.Tip.warn("请注意填写必填项");


MelUi.Tip.error("接口调用失败");


MelUi.Tip.info("系统将于今晚维护");


MelUi.Tip({
    type: "error",
    message: "请求异常，请稍后重试",
    duration: 5000,
    position: "bottom"
});


MelUi.Tip.info("第一条消息");
MelUi.Tip.success("第二条消息");
MelUi.Tip.warn("第三条消息");


MelUi.Tip({
    type:"info",
    message:"带遮罩的Tip，点击右上角×关闭",
    mask:true,
    duration:0
});
3、Toast轻提示（天猫风格白底提示）
调用方式说明

//（1）快速调用：直接传入字符串
// 成功 MelUi.Toast.success("消息")
// 警告 MelUi.Toast.warn("消息")
// 信息 MelUi.Toast.info("消息")
// 错误 MelUi.Toast.error("消息")

//（2）完全调用：传入object参数
MelUi.Toast({
    type: "success", // success成功 / warn警告 / info信息 / error错误
    message: "已加入购物车", // 提示文本内容
    duration: 2500, // 自动关闭延时(毫秒)，0常驻
    position: "center" // 弹出位置 center / top / bottom
});


MelUi.Toast.success("已加入购物车");


MelUi.Toast.warn("库存即将不足");


MelUi.Toast.error("提交表单失败");


MelUi.Toast.info("操作正在处理");


MelUi.Toast({
    type:"error",
    message:"提交失败，请重试",
    duration:3500
});

melui.alert({

melui.alert({

    title: "提示",

5、Confirm确认对话框
有标题confirm

melui.confirm({
    title: "删除确认",
    message: "数据删除后无法恢复，确认继续吗？",
    theme: "danger",
    onClick: function(ret) {
        if(ret === 2) {
            melui.alert({ message: "确认删除" });
        }
    }
});
无标题confirm

melui.confirm({
    message: "确定执行该操作吗？"
});
自定义按钮名称

melui.confirm({
    title: "操作选择",
    message: "请确认操作",
    buttons: [
        { text: "返回", type: "default" },
        { text: "执行", type: "warning" }
    ]
});
按钮竖向排列 vert

melui.confirm({
    title: "竖向排布",
    message: "按钮垂直排列",
    btnDirection: "vert"
});
按钮通栏铺满 btnFullWidth

melui.confirm({
    title: "通栏按钮",
    message: "按钮铺满整行",
    btnFullWidth: true
});
6、prompt输入弹窗（单行、多行textarea、带单位unit）
单行输入，带单位 元

melui.prompt({
    title: "输入金额",
    prompt: {
        inputType: "input",
        placeholder: "请输入金额",
        defaultValue: "100",
        unit: "元"
    },
    onClick: function(ret, val) {
        if(ret === 2) {
            melui.alert({ message: "输入：" + val });
        }
    }
});
textarea多行输入

melui.prompt({
    title: "备注信息",
    prompt: {
        inputType: "textarea",
        placeholder: "请输入备注内容"
    }
});
7、自定义HTML内容弹窗

melui.dialog({
    title: "自定义HTML内容",
    dangerouslyUseHTMLString: true,
    content: '<div><p>这里是html片段</p><p style="color:red">红色文字内容</p></div>'
});


melui.dialog({
    title: "自定义宽度",
    dangerouslyUseHTMLString: true,
    content: '<p>弹窗宽度550px</p>',
    width: "550px"
});


melui.dialog({
    title: "全屏弹窗",
    dangerouslyUseHTMLString: true,
    content: '<p>这是全屏弹窗内容</p>',
    fullScreen: true
});


melui.dialog({
    title: "固定高度滚动",
    dangerouslyUseHTMLString: true,
    content: '<p>测试内容</p><p>测试内容</p><p>测试内容</p><p>测试内容</p><p>测试内容</p><p>测试内容</p>',
    adaptive: true,
    height: "300px"
});
安全警告：dangerouslyUseHTMLString开启时content直接渲染HTML，不要渲染不受信任的外部输入，防止XSS攻击。
8、特殊回调与特性演示

melui.prompt({
    title: "onOpen自动聚焦",
    prompt: {
        inputType:"input",
        placeholder:"自动获得焦点"
    },
    onOpen: function(ins) {
        if(ins.inputEl) ins.inputEl.focus();
    }
});


melui.alert({

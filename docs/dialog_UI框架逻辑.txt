
简单弹窗
MelUI.alert = function(content, title = '提示') {
  // 预先固定参数，调用底层Dialog
  return new Promise((resolve) => {
    const dialog = new MelUI.Dialog({
      title: title,
      content: 带上信息图标 + 文字内容,
      confirmText: '确定',
      cancelText: '', // 无取消按钮
      showClose: false,
      closeOnMask: false,
      onConfirm: () => {
        dialog.close();
        resolve();
      }
    })
    dialog.open();
  })
}


用户使用：
// 极简调用
MelUI.alert('保存成功');
// 搭配await
await MelUI.alert('提交完毕');


确认弹窗
MelUI.confirm = function(options) {
  return new Promise((resolve, reject) => {
    const dialog = new MelUI.Dialog({
      title: options.title,
      content: 警示图标+文字,
      confirmText: '确定',
      cancelText: '取消',
      closeOnMask: false,
      onConfirm: () => {
        dialog.close();
        resolve();
      },
      onCancel: () => {
        dialog.close();
        reject();
      }
    })
    dialog.open();
  })
}

用户使用
MelUI.confirm('确定删除数据？删除不可恢复')
.then(()=>{ 删除接口 })
.catch(()=>{ 取消操作 })

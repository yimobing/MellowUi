## 阶段3：编写 util入口 `src/util/util.js`

```
// 1.IE环境简易检测警告，仅warn，不阻断执行
const isIE = !!window.ActiveXObject || "ActiveXObject" in window;
if(isIE){
  const testEl = document.createElement('div');
  if(!testEl.classList){
    console.warn('[mel‑util warn]: IE浏览器检测到原型补丁缺失，请确认已引入 melui.min.js，请勿使用自行裁剪过的core版本。');
  }
}

// 导入core的对象合并工具
const { mergeObj } = window.melui;

// 导入各个子模块
import * as base from './base/base.js';
import * as dom from './dom.js';
import * as string from './string.js';
import * as array from './array.js';
import * as date from './date.js';
import * as format from './format.js';
import * as validate from './validate.js';
import * as random from './random.js';
import * as biz from './biz.js';
import * as widget from './widget.js';

// 组装命名空间
const utilNs = {
  base,
  dom,
  string,
  array,
  date,
  format,
  validate,
  random,
  biz,
  widget
};

// 挂载全局melui.util
window.melui = window.melui || {};
window.melui.util = utilNs;

export { utilNs };
```

> 
> rollup打包util入口指向 `src/util/util.js`，external设置`["melui"]`；产出：`mel‑util.js` / `mel‑util.min.js`。
